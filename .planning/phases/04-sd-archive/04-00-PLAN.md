    override fun toString() = "LinkIoCtrlReply(type=$type, bytes=${payload.size})"
    companion object {
        // Shared never-emitting flow for links without IOCTRL replies (default member body).
        val NONE: SharedFlow<LinkIoCtrlReply> = MutableSharedFlow()

}
