  <behavior>
    QualityController(switchTimeoutMs = 8_000):
    - target(LAN) == HD, target(P2P) == HD, target(RELAY|TCP_RELAY|SDEV_RELAY|UNKNOWN) == SD; after setManual(SD) target(LAN) == SD (manual wins for the screen lifetime).
    - observe(KeyFrameObservation(uc, 1920, 1080)) -> state.observed HD; width 640 -> SD.
    - onNewLink() clears pending and the per-link "auto request done" flag; keeps manual and switchSupported.
    - needsAutoRequest(mode): false before the first observed key frame of the link; true once when observed != target(mode) and switchSupported; false after beginRequest was called on this link.
    - beginRequest(HD, useCount = 7, nowMs = 1000) -> state.pending HD; observe(useCount 7, width 640) -> pending cleared (useCount match), observed SD (shown = observed); observe(useCount 3, width 1920) while pending HD -> pending cleared (width class match).
    - beginRequest(HD, useCount = null, nowMs) -> switchSupported false, pending null.
    - tick(nowMs = 9000) with pending since 1000 -> pending null, switchSupported false; tick(8999) -> still pending.
    - onUnsupported() -> switchSupported false, pending null.
    - setManual(q) returns true when switchSupported and q != observed (caller should send), false otherwise; setManual is ignored (returns false, manual unchanged) when switchSupported is false.
    ResumePolicy(windowMs = RESUME_WINDOW_MS = 300_000):
    - onStopped("background", wasActive = true, nowMs = 0); shouldAutoResume(300_000) == true; then a second call == false (consumed).
    - shouldAutoResume(300_001) == false.
    - onStopped("user" | "screen-left", true, 0) -> false; onStopped("background", wasActive = false, 0) -> false.
    - onStarted() clears any pending resume (shouldAutoResume == false afterwards).
  </behavior>

class QualityController(private val switchTimeoutMs: Long = SWITCH_TIMEOUT_MS) {
    val state: QualityState get()
    fun target(mode: LinkMode): Quality
    fun onNewLink()
    fun observe(frame: KeyFrameObservation)
    fun needsAutoRequest(mode: LinkMode): Boolean
    fun beginRequest(quality: Quality, useCount: Int?, nowMs: Long)
    fun tick(nowMs: Long)
    fun onUnsupported()
    fun setManual(quality: Quality): Boolean
    companion object { const val SWITCH_TIMEOUT_MS = 8_000L }
}
class ResumePolicy(private val windowMs: Long = RESUME_WINDOW_MS) {
    fun onStopped(reason: String, wasActive: Boolean, nowMs: Long)
    fun onStarted()
    fun shouldAutoResume(nowMs: Long): Boolean
    companion object { const val RESUME_WINDOW_MS = 300_000L; const val REASON_BACKGROUND = "background" }
}
