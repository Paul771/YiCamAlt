```kotlin
class LiveSessionSupervisor(
    private val deviceId: String,
    private val credentials: P2pCredentialsSource,
    private val factory: CameraLinkFactory,
    private val sessions: MediaSessionFactory,
    private val network: NetworkMonitor,
    private val scope: CoroutineScope,
    private val ioDispatcher: CoroutineDispatcher,
    private val now: () -> Long,
    private val timing: Timing = Timing(),
    private val quality: QualityController = QualityController(timing.qualitySwitchMs),
    private val resume: ResumePolicy = ResumePolicy(timing.resumeWindowMs),
) {
    data class Timing(
        val networkDelaysMs: List<Long> = listOf(2_000, 5_000, 10_000, 20_000),
        val maxAttempts: Int = 5,
        val wrongPasswordDelayMs: Long = 2_000,
        val kickDelayMs: Long = 10_000,
        val stallMs: Long = 10_000,
        val healthyResetMs: Long = 30_000,
        val drainMs: Long = 5_000,
        val networkChangeDebounceMs: Long = 1_500,
        val tickMs: Long = 1_000,
        val qualitySwitchMs: Long = 8_000,
        val resumeWindowMs: Long = 300_000,
    )
    val state: StateFlow<SupervisorState>
    val qualityState: StateFlow<QualityState>
    val currentLinkState: StateFlow<LinkState>       // last link's state for the debug row
    fun start()                                      // fresh run: counters reset, resume.onStarted()
    fun retry() = start()
    fun stop(reason: String)                         // "user" | "screen-left" | "background"
    fun onForeground(): Boolean                      // resume.shouldAutoResume(now()) -> start()
    fun setQuality(q: Quality)
}
fun classify(error: LinkError): ErrorClass
fun classifyClosed(reason: String): ErrorClass?
```
Run loop (one `Job` launched in `scope`, replaced on start(), cancelled on stop()):
  1. if `!network.state.value.available` -> state NoNetwork, `network.state.first { it.available }`, attempt = 1.
  2. drain: `previous?.let { withTimeoutOrNull(drainMs) { it.state.first { s -> s is Failed || s is Closed } } }` (log BLOCK_DRAIN waitedMs, result=terminal|timeout).
  3. creds = refreshNext ? `withContext(io) { credentials.refreshCredentials(deviceId) }` : `credentials.credentialsFor(deviceId)`; null -> Fatal(LinkError(NO_CREDENTIALS, 0)).
  4. state Connecting(attempt); session = sessions.create(creds); link = `withContext(io) { factory.open(creds, LinkOptions()) }`; if the run was stopped meanwhile -> `link.close(stopReason)` on io + session.release(), return. `quality.onNewLink()`; `link.setAudioSink(session.audioSink)`; `link.start(session.videoSink)`; log BLOCK_OPEN_LINK attempt.
  5. watch (coroutineScope): collect link.state (Streaming -> Live, record streamingSince, key frames via ticker), ticker every tickMs (stall: framesIn unchanged since lastGrowthAt >= stallMs -> close "stall"; healthy reset at healthyResetMs; quality: observe(session.lastKeyFrame()) when changed, needsAutoRequest -> request, quality.tick(now())); collect link.capabilities (resolutionSwitch == false -> quality.onUnsupported()); collect network (generation change while Live -> delay(debounce) -> close "network-change"). Ends when link.state is terminal. All closes initiated by the supervisor go through `closeLink(link, reason)` = `scope.launch(io) { link.close(reason) }` and remember `ownCloseReason`.
  6. session.release(); previous = link; outcome -> cls (Failed -> classify; Closed(reason) -> ownCloseReason ?: reason -> classifyClosed). null -> return.
  7. special case first: own close reason "network-change" -> attempt = 1 and reconnect at once (no Waiting delay). Otherwise per class as in behavior; waits use `withTimeoutOrNull(delay) { network.state.first { !it.available } }` so a lost network during Waiting jumps to step 1 without consuming attempts.
stop(reason): set stopReason, cancel the job, close the current link with reason (io), release the session, `resume.onStopped(reason, wasActive = state is Connecting|Waiting|Live|NoNetwork, now())`, state Stopped(reason), log BLOCK_STOP.
LiveFakes.kt (test): `FakeCameraLink(deviceId)` (MutableStateFlow state, `startCalls`, `closeReasons`, `audioSinkSetBeforeStart`, `requestResolution` returns a scripted Int? and records resolutions, `capabilitiesFlow`, `terminalOnClose = true` toggle), `FakeLinkFactory` (queue of links or a lambda; records open times via a `now` lambda and credentials), `FakeNetworkMonitor` (MutableStateFlow<NetState>), `FakeMediaSession` (var frames, var keyFrame, released flag), `FakeCredentialsSource` (counts credentialsFor/refreshCredentials, returns scripted P2pCredentials with synthetic values).
