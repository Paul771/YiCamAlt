  <behavior>
    AdtsParser:
    - Synthetic header for AAC-LC (profile bits 01), freqIndex 8 (16000 Hz), channels 1, protection_absent 1, frameLen 7+N -> AdtsFrame(objectType 2, freqIndex 8, channels 1, headerLen 7, frameLen 7+N); audioSpecificConfig -> [0x14, 0x08]; sampleRate(8) -> 16000.
    - protection_absent 0 -> headerLen 9.
    - Returns null for: < 7 bytes; first byte != 0xFF; layer bits != 0; freqIndex > 12; number_of_raw_data_blocks != 0; frameLen < headerLen; off + frameLen > size.
    - parseAll(bytes) walks consecutive frames (two frames back to back -> 2 results); stops at the first invalid header.
    AudioPipeline:
    - muted = true by default: a valid unit increments unitsIn and framesOk, AacFrameSink gets nothing.
    - muted = false: each ADTS frame -> onAacFrame(config, raw frame without header, ptsUs) with config.sampleRate 16000, channels 1, csd0 [0x14,0x08]; ptsUs strictly increasing.
    - encrypted = true + password "SYNTHpw1234567" (15 chars -> 16-byte key): unit built by AES-ECB-encrypting every full 16-byte block of a valid ADTS payload (tail of 5 bytes left plain) is decoded to the same frames; decryptOk = true.
    - encrypted = true with password null or key length not 16/24/32 -> payload used as is (fail-open), decryptOk stays null.
    - codecId 140 or any id != 138 -> unit counted in badUnits, codecId recorded, availability(...) == false.
    - availability(streamingAtMs = 1000, nowMs = 6999) == null with no valid frame; (nowMs = 7000) == false; after one valid frame == true regardless of time; streamingAtMs null -> null.
    - Garbage unit (no ADTS sync after decrypt) -> badUnits++; when the unit was encrypted, decryptOk = false (true again after a later valid encrypted unit).
    - No dropping inside the pipeline: 10 valid frames with muted = false -> exactly 10 onAacFrame calls (queue limits are the decoder's job, plan 03-07).
    - toString of AudioStats has no bytes.
  </behavior>

data class AdtsFrame(val objectType: Int, val freqIndex: Int, val channels: Int, val headerLen: Int, val frameLen: Int)
object AdtsParser {
    val SAMPLE_RATES = intArrayOf(96000, 88200, 64000, 48000, 44100, 32000, 24000, 22050, 16000, 12000, 11025, 8000, 7350)
    fun parse(b: ByteArray, off: Int): AdtsFrame?            // exactly the research "ADTS -> csd-0" checks
    fun parseAll(b: ByteArray): List<AdtsFrame>              // frames at consecutive offsets
    fun audioSpecificConfig(f: AdtsFrame): ByteArray         // (objType shl 11) or (freqIndex shl 7) or (channels shl 3), 2 bytes
    fun sampleRate(freqIndex: Int): Int
}
// AudioPipeline.kt
class AacConfig(val objectType: Int, val sampleRate: Int, val channels: Int, val csd0: ByteArray) { equals/hashCode on values incl. csd0.contentEquals; toString without csd bytes }
fun interface AacFrameSink { fun onAacFrame(config: AacConfig, frame: ByteArray, ptsUs: Long) }
data class AudioStats(val unitsIn: Long, val framesOk: Long, val badUnits: Long, val codecId: Int?, val sampleRate: Int?, val channels: Int?, val decryptOk: Boolean?)
class AudioPipeline(private val password: String?, private val out: AacFrameSink, private val nanoTime: () -> Long = System::nanoTime) : AudioUnitSink {
    @Volatile var muted: Boolean = true
    @Synchronized override fun onAudioUnit(unit: AudioUnit)
    fun stats(): AudioStats
    fun availability(streamingAtMs: Long?, nowMs: Long): Boolean?   // AUDIO_PROBE_MS = 6_000
    companion object { const val AUDIO_PROBE_MS = 6_000L }
}
