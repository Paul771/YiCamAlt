// app/src/main/java/com/yicamalt/stream/StreamTypes.kt   (package com.yicamalt.stream, ROLE: TYPES)
enum class VideoCodec(val wireId: Int, val mime: String) {
    H264(78, "video/avc"), H265(81, "video/hevc");
    companion object { fun fromWireId(id: Int): VideoCodec? }
}
/** 24-byte frame head (TNP big-endian, TUTK little-endian). Parsed by FrameHeadCodec (plan 02-04). */
data class FrameInfo(
    val codecId: Int, val flags: Int, val liveFlag: Int, val onlineNum: Int, val useCount: Int,
    val seq: Int, val width: Int, val height: Int, val timestampSec: Long,
    val isDay: Int, val cover: Int, val outLoss: Int, val inLoss: Int, val timestampMs: Long,
) { val isKeyFrame: Boolean get() = (flags and 1) == 1 }
/** One video unit as read from the transport: head + payload after the 24-byte head (I-frames still encrypted). */
class VideoUnit(val channel: Int, val info: FrameInfo, val payload: ByteArray) {
    override fun toString() = "VideoUnit(ch=$channel, codec=${info.codecId}, key=${info.isKeyFrame}, seq=${info.seq}, bytes=${payload.size})"
}
/** Decrypted, Annex-B access unit ready for MediaCodec. csd: AVC [SPS, PPS] / HEVC [VPS+SPS+PPS], each with start code. */
class DecodableFrame(val codec: VideoCodec, val isKeyFrame: Boolean, val annexB: ByteArray, val csd: List<ByteArray>,
                     val ptsUs: Long, val width: Int, val height: Int) {
    override fun toString() = "DecodableFrame(codec=$codec, key=$isKeyFrame, bytes=${annexB.size}, csd=${csd.size}, ${width}x$height)"
}
fun interface VideoUnitSink { fun onVideoUnit(unit: VideoUnit) }
interface DecodableFrameSink {
    fun onFrame(frame: DecodableFrame)
    fun onUnsupportedCodec(codecId: Int)
}

// app/src/main/java/com/yicamalt/stream/CameraLink.kt   (package com.yicamalt.stream, ROLE: TYPES)
enum class LinkMode { P2P, RELAY, TCP_RELAY, SDEV_RELAY, LAN, UNKNOWN }   // TNP bMode 0..3; TUTK Mode 2 = LAN
data class LinkInfo(val transport: TransportType, val mode: LinkMode, val lan: Boolean, val connectMs: Int)
enum class LinkErrorKind { NATIVE_UNAVAILABLE, NO_CREDENTIALS, UNSUPPORTED_TRANSPORT, ALREADY_OPEN, OFFLINE, CREDENTIALS,
    NETWORK, BUSY, WRONG_PASSWORD, REMOTE_CLOSED, KICKED, TIMEOUT, PROTOCOL, INTERNAL }
data class LinkError(val kind: LinkErrorKind, val code: Int)   // code = vendor/app code (e.g. -3006), 0 when none
sealed interface LinkState {
    object Idle : LinkState
    object Connecting : LinkState
    data class Connected(val info: LinkInfo) : LinkState                       // session up, burst not yet answered
    data class Streaming(val info: LinkInfo, val authOk: Boolean) : LinkState  // first video unit received
    data class Failed(val error: LinkError) : LinkState                         // terminal; no auto-retry
    data class Closed(val reason: String) : LinkState                           // terminal; reason in {user, kicked, screen-left, background, error, timeout} (error/timeout normally end in Failed; the set is shared with the BLOCK_CLOSE log reason)
}
data class LinkOptions(val forceRelay: Boolean = false)
interface CameraLink {
    val deviceId: String
    val state: kotlinx.coroutines.flow.StateFlow<LinkState>
    /** Non-blocking: spawns the connect thread; units go to [sink] from reader threads. Second call is ignored. */
    fun start(sink: VideoUnitSink)
    /** Idempotent, callable from any thread, never blocks longer than ~2 s (abort + bounded join). */
    fun close(reason: String = "user")
}
interface CameraLinkFactory {
    /** One session per camera: when a link for credentials.deviceId is open, returns a link already in Failed(ALREADY_OPEN). */
    fun open(credentials: P2pCredentials, options: LinkOptions): CameraLink
}
fun interface P2pCredentialsSource { fun credentialsFor(deviceId: String): P2pCredentials? }

/** The only seam between our code and the vendor PPPP library. Android impl: AndroidPpppNative (plan 02-03); JVM: FakePpppNative. */
interface PpppNative {

    fun configDebug(enable: Int, level: Int): Int
    fun apiVersion(): Int

    fun connect(did: String, flag: Byte, wakeup: Boolean, serverString: String, licenseKey: String): Int
    fun connectBreak(did: String): Int
    fun check(handle: Int): PpppCheck
    /** Reads into buf[0 until size[0]]; size[0] is updated to the bytes actually read, also on -3003 (TIME_OUT). */
    fun read(handle: Int, channel: Int, buf: ByteArray, size: IntArray, timeoutMs: Int): Int
    fun write(handle: Int, channel: Int, buf: ByteArray, size: Int): Int
    fun forceClose(handle: Int): Int
}
/** Value-free projection of st_PPPP_Session: remote address is reduced to a private/public boolean in-process. */

    const val NO_RELAY_SERVER = -3010; const val INVALID_SESSION_HANDLE = -3011; const val SESSION_CLOSED_REMOTE = -3012
    const val SESSION_CLOSED_TIMEOUT = -3013; const val SESSION_CLOSED_CALLED = -3014; const val REMOTE_SITE_BUFFER_FULL = -3015
    const val MAX_SESSION = -3017; const val USER_CONNECT_BREAK = -3019; const val INVALID_APILICENSE = -3021
    const val INVALID_SERVER_STRING = -3023; const val NO_AVAILABLE_P2P_SERVER = -3026; const val DEVICE_MAX_SESSION = -3029
    const val APP_KICKED = -3100; const val APP_KICK_MAX_SESSION = -3101; const val APP_WRONG_PASSWORD = -20009; const val APP_MANUAL_RETRY = -9000

}
